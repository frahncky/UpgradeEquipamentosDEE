% Selo do nível de prioridade, com cor por faixa.
\newcommand{\seloPrioridade}[1]{%
  \ifcsdef{prioridade@cor@#1}{\color{\csuse{prioridade@cor@#1}}}{\color{cinzamedio}}%
  \footnotesize\sffamily\bfseries\rotuloPrioridade{#1}%
}
\csdef{prioridade@cor@P1}{ifmavermelho}
\csdef{prioridade@cor@P2}{ambar}
\csdef{prioridade@cor@P3}{cinzamedio}

% Uma linha do diretório a partir da chave do registro.
\newcommand{\linhaContato}[1]{%
  \stepcounter{contatoNum}%
  \thecontatoNum &
  {\bfseries\csuse{contato@#1@empresa}}\par
  {\seloPrioridade{\csuse{contato@#1@nivel}}}\par
  {\footnotesize\color{cinzamedio}\texttt{\detokenize{#1}} — \csuse{contato@#1@status}} &
  {\bfseries\csuse{contato@#1@setor}}\par
  \vspace{0.6mm}\csuse{contato@#1@canal}\par
  \vspace{0.6mm}{\footnotesize\color{cinzaescuro}\textbf{Estratégia:} \csuse{contato@#1@estrategia}}\\
  \addlinespace[1.2mm]
}
